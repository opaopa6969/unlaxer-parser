fn map_associative_$NAME(tree: &Tree, capture: &unlaxer_runtime::Capture) -> Result<Ast, String> {

    if values.len() > 1 { return Err(format!("expected one associative operand at CP {}:{}", capture.span.start, capture.span.end)); }
    if let Some(value) = values.pop() { return Ok(value); }
    let literal = unlaxer_runtime::java_capture_text(tree.text(capture.span)?).to_owned();

}
