
/// One independently owned mapping snapshot. Selection never reparses or changes it.
#[derive(Debug, Clone)]
pub struct MappedTree {
    source: String,
    offsets: Vec<usize>,
    candidates: Vec<MappingCandidate>,
}
#[derive(Debug, Clone)]
pub struct MappingCandidate {
    node: usize,
    depth: usize,
    span: unlaxer_runtime::Span,
    simple_name: &'static str,
    ast: Ast,
}
impl MappingCandidate {
    pub fn node_id(&self) -> usize { self.node }
    pub fn depth(&self) -> usize { self.depth }
    pub fn span(&self) -> unlaxer_runtime::Span { self.span }
    pub fn simple_name(&self) -> &'static str { self.simple_name }
    pub fn ast(&self) -> &Ast { &self.ast }
}
#[derive(Debug, Clone, Copy)]
pub struct MappingLimits {
    pub candidates: usize,
    pub projection_work: usize,
}
impl Default for MappingLimits {
    fn default() -> Self { Self { candidates: 4096, projection_work: 1_000_000 } }
}
impl MappedTree {
    pub fn source(&self) -> &str { &self.source }
    pub fn source_text(&self, span: unlaxer_runtime::Span) -> Option<&str> {
        if span.start > span.end { return None; }
        self.source.get(*self.offsets.get(span.start)?..*self.offsets.get(span.end)?)
    }
    pub fn candidates(&self) -> &[MappingCandidate] { &self.candidates }
    pub fn select_default(&self) -> &MappingCandidate { self.select(None) }
    /// Preferred simple name, then shallower Java token depth, later start, later preorder.
    /// Missing names fall back to the default; blank follows Java String.isBlank.
    pub fn select(&self, preferred: Option<&str>) -> &MappingCandidate {
        let preferred = preferred.filter(|name| !name.chars().all(|c| matches!(c,
            '\u{9}'..='\u{d}' | '\u{1c}'..='\u{20}' | '\u{1680}' |
            '\u{2000}'..='\u{2006}' | '\u{2008}'..='\u{200a}' | '\u{2028}' | '\u{2029}' |
            '\u{205f}' | '\u{3000}')));
        let mut best = &self.candidates[0];
        for candidate in &self.candidates[1..] {
            let favored = preferred.is_none_or(|name| candidate.simple_name == name);
            let old_favored = preferred.is_none_or(|name| best.simple_name == name);
            if favored && !old_favored || favored == old_favored &&
                (candidate.depth < best.depth || candidate.depth == best.depth && candidate.span.start >= best.span.start) {
                best = candidate;
            }
        }
        best
    }
}
/// Normal parse_tree CSTs intentionally lack the metadata required for legacy Java depth order.
/// Use parser::parse_tree_for_mapping or parse_mapped_tree.
pub fn map_parsed_tree(tree: &Tree) -> Result<MappedTree, String> {
    if __PREFERRED_STRICT_ROOT && tree.nodes.get(tree.root).is_none_or(|node| node.rule != __PREFERRED_ROOT) {
        return Err("E-PREFERRED: committed grammar root required".into());
    }
    map_subtree_tree_with_limits(tree, tree.root, MappingLimits::default())
}
pub fn map_subtree_tree(tree: &Tree, root: usize) -> Result<MappedTree, String> {
    map_subtree_tree_with_limits(tree, root, MappingLimits::default())
}
pub fn parse_mapped_tree(source: &str) -> Result<MappedTree, String> {
    let tree = super::parser::parse_tree_for_mapping(source).map_err(|error| error.to_string())?;
    map_parsed_tree(&tree)
}
pub fn map_subtree_tree_with_limits(tree: &Tree, root: usize, limits: MappingLimits) -> Result<MappedTree, String> {
    tree.validate_structure().map_err(str::to_owned)?;
    let entry = tree.nodes.get(root).ok_or("E-PREFERRED: invalid subtree")?;
    if entry.rule >= __PREFERRED_RULE_COUNT || !entry.captures.iter().any(|capture| capture.name == __PREFERRED_MARKER) {
        return Err("E-PREFERRED: mapping metadata for this grammar required".into());
    }
    let mut work = vec![0usize; tree.nodes.len()];
    let mut heights = vec![0usize; tree.nodes.len()];
    let mut recovered = vec![false; tree.nodes.len()];
    for (id, node) in tree.nodes.iter().enumerate() {
        let mut edges = std::collections::BTreeMap::<usize, (usize, usize)>::new();
        for &child in &node.children { edges.entry(child).or_default().0 += 1; }
        for &child in node.captures.iter().filter(|c| !c.name.starts_with("#preferred-")).flat_map(|c| &c.nodes) {
            edges.entry(child).or_default().1 += 1;
        }
        heights[id] = 1 + edges.keys().map(|&child| heights[child]).max().unwrap_or(0);
        recovered[id] = node.rule == unlaxer_runtime::RECOVERY_ERROR_RULE || edges.keys().any(|&child| recovered[child]);
        work[id] = edges.into_iter().fold(1usize, |sum, (child, (structural, captured))|
            sum.saturating_add(work[child].saturating_mul(structural.max(captured))));
    }
    if recovered[root] { return Err("cannot map recovered syntax".into()); }
    // The existing recursive AST projection must also be safe for caller-mutated arenas.
    if work[root] > limits.projection_work { return Err("E-PREFERRED: traversal work limit".into()); }
    if heights[root] > 256 { return Err("E-PREFERRED: projection depth limit".into()); }
    let mut remaining = limits.projection_work;
    let mut pending = vec![(root, usize::from(__preferred_recovery_wrapper(entry.rule)))];
    let mut candidates = Vec::new();
    while let Some((id, depth)) = pending.pop() {
        if depth > 4096 { return Err("E-PREFERRED: token depth limit".into()); }
        let node = &tree.nodes[id];
        if __preferred_skipped(node.rule) { continue; }
        if __preferred_candidate(node.rule) {
            if candidates.len() >= limits.candidates || work[id] > remaining {
                return Err("E-PREFERRED: mapping snapshot limit".into());
            }
            remaining -= work[id];
            let ast = required(map_node(tree, id)?, "mapping candidate")?;
            candidates.push(MappingCandidate { node:id, depth, span:node.span, simple_name:__preferred_name(&ast), ast });
        }
        // Synthetic value boundaries do not exist as generated Java rule tokens.
        // Index capture edges once: a list rule may contain thousands of call sites.
        let mut weights = std::collections::BTreeMap::<usize, Vec<usize>>::new();
        let mut extras = std::collections::BTreeMap::<usize, usize>::new();
        for capture in &node.captures {
            if let Some(value) = capture.name.strip_prefix("#preferred-depth:") {
                let weight = value.parse::<usize>().map_err(|_| "E-PREFERRED: invalid depth")?;
                if !(1..=4096).contains(&weight) { return Err("E-PREFERRED: invalid depth".into()); }
                for &child in &capture.nodes { weights.entry(child).or_default().push(weight); }
            } else if capture.name == "#preferred-extra:2" {
                for &child in &capture.nodes { *extras.entry(child).or_default() += 2; }
            }
        }
        let mut children = Vec::new();
        let mut scan: Vec<_> = node.children.iter().rev().map(|&child| (child, 0usize)).collect();
        while let Some((child, inherited)) = scan.pop() {
            let child_node = &tree.nodes[child];
            let extra = extras.get(&child).copied().unwrap_or(0);
            let inherited = inherited.saturating_add(extra);
            if child_node.rule >= __PREFERRED_RULE_COUNT {
                scan.extend(child_node.children.iter().rev().map(|&id| (id, inherited)));
            } else {
                let Some(values) = weights.get(&child) else { return Err("E-PREFERRED: missing call-site metadata".into()); };
                if values.len() != 1 { return Err("E-PREFERRED: ambiguous call-site metadata".into()); }
                let weight = values[0];
                children.push((child, depth.saturating_add(weight).saturating_add(inherited).saturating_add(usize::from(__preferred_recovery_wrapper(child_node.rule)))));
            }
        }
        pending.extend(children.into_iter().rev());
    }
    if candidates.is_empty() { return Err("No mapped node found in token tree".into()); }
    let offsets = tree.source.char_indices().map(|(i,_)| i).chain(std::iter::once(tree.source.len())).collect();
    Ok(MappedTree { source:tree.source.clone(), offsets, candidates })
}
