mod generated;
use generated::{ast::{Ast, AstValue}, evaluator::{evaluate, Semantics}};
use unlaxer_runtime::{Span, json_string};
use std::io::{self, BufRead};

struct Values;
impl Values {
    fn value(&mut self, value: &AstValue) -> String {
        match value {
            AstValue::Text { text, .. } => format!("T:{text}"),
            AstValue::Node(node) => evaluate(node, self),
        }
    }
}
impl Semantics for Values {
    type Output = String;
    fn eval_root(&mut self, head: &AstValue, maybe: Option<&AstValue>, items: &[AstValue], _: Span) -> String {
        let head = self.value(head);
        let maybe = maybe.map(|value| self.value(value)).unwrap_or_else(|| "-".into());
        let items = items.iter().map(|value| self.value(value)).collect::<Vec<_>>().join(",");
        format!("{head}|{maybe}|{items}")
    }
    fn eval_leaf(&mut self, value: &str, _: Span) -> String { format!("L:{value}") }
    fn eval_other_leaf(&mut self, value: &str, _: Span) -> String { format!("O:{value}") }
}

fn text_spans(ast: &Ast) -> String {
    let Ast::Root { head, maybe, items, .. } = ast else { panic!("root expected") };
    let values = std::iter::once(head).chain(maybe.iter()).chain(items.iter());
    let texts = values.filter_map(|value| match value {
        AstValue::Text { text, span } => Some(format!("[{},{},{}]", span.start, span.end, json_string(text))),
        AstValue::Node(_) => None,
    }).collect::<Vec<_>>().join(",");
    format!("[{texts}]")
}

fn main() {
    for line in io::stdin().lock().lines() {
        let encoded = line.unwrap();
        let bytes = (0..encoded.len()).step_by(2)
            .map(|i| u8::from_str_radix(&encoded[i..i+2], 16).unwrap()).collect();
        let input = String::from_utf8(bytes).unwrap();
        let mut context = unlaxer_runtime::ParseContext::new(&input);
        let ok = generated::parser::parse_context(&mut context).is_ok();
        print!(r#"{{"prefix":[{},{},{}],"ast":"#, ok, context.position(), context.matched_position());
        match generated::parser::parse_tree(&input) {
            Ok(tree) => {
                let ast = generated::mapper::map(&tree).unwrap();
                drop(tree);
                drop(context);
                drop(input);
                print!(r#"{},"value":{},"texts":{}"#, ast.canonical_json(), json_string(&evaluate(&ast, &mut Values)), text_spans(&ast));
            }
            Err(_) => print!("null"),
        }
        println!("}}");
    }
}
