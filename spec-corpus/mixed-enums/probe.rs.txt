mod generated;
use generated::{ast::{Ast,AstValue},evaluator::{evaluate,Semantics}};
use unlaxer_runtime::{json_string,Span,ParseOptions,Diagnostics,Memoization};
use std::io::{self,BufRead};
struct Observe;
fn value(item:&AstValue) -> String {
    let span=item.span();
    match item {
        AstValue::Text{text,..} => format!(r#"["text",{},{},{}]"#,json_string(text),span.start,span.end),
        AstValue::Node(node) => evaluate(node,&mut Observe),
        AstValue::EnumMode{value,..} => format!(r#"["enum","Mode",{},{},{}]"#,json_string(value.text()),span.start,span.end),
        AstValue::EnumHeat{value,..} => format!(r#"["enum","Heat",{},{},{}]"#,json_string(value.text()),span.start,span.end),
    }
}
impl Semantics for Observe {
    type Output=String;
    fn eval_values(&mut self,head:Option<&AstValue>,values:&[AstValue],_:Span)->String {
        format!(r#"{{"head":{},"values":[{}]}}"#,head.map(value).unwrap_or("null".into()),values.iter().map(value).collect::<Vec<_>>().join(","))
    }
    fn eval_leaf(&mut self,text:&str,span:Span)->String {
        format!(r#"["node","Leaf",{},{},{}]"#,json_string(text),span.start,span.end)
    }
}
fn main() {
    for line in io::stdin().lock().lines() {
        let encoded=line.unwrap();
        let input=String::from_utf8((0..encoded.len()).step_by(2).map(|i|u8::from_str_radix(&encoded[i..i+2],16).unwrap()).collect()).unwrap();
        for memo in [Memoization::Off,Memoization::SafeFailures] {
        for diagnostics in [Diagnostics::Auto,Diagnostics::Detailed,Diagnostics::DetailedOnFailure] {
        match generated::parser::parse_tree_with_options(&input,ParseOptions::with_memoization(memo).with_diagnostics(diagnostics)) {
            Ok(tree)=>{let ast=generated::mapper::map(&tree).unwrap();drop(tree);let result=evaluate(&ast,&mut Observe);if matches!(ast,Ast::Leaf{..}) {println!("{}",with_dap(format!(r#"{{"root":{}}}"#,result),&input));}else{println!("{}",with_dap(result,&input));}},
            Err(_)=>println!(r#"{{"failure":true}}"#),
        }
        }
        }
    }
}
